\chapter{Generative Models}
\label{chap:generative_models}

\begin{tldr}
Generative models learn to produce new data samples from a learned distribution. This chapter covers the three major paradigms: Variational Autoencoders (VAEs), Generative Adversarial Networks (GANs), and Diffusion Models. Understanding the trade-offs (sample quality vs. diversity vs. training stability) and when to use each approach is essential for modern ML roles.
\end{tldr}

\section{Generative Model Taxonomy}
\label{sec:gen_taxonomy}

\subsection{The Generative Modeling Problem}

Given samples $\{x_1, \ldots, x_n\}$ from unknown distribution $p_{data}(x)$, learn a model $p_\theta(x)$ that:
\begin{enumerate}
    \item Approximates $p_{data}(x)$ well
    \item Can generate new samples $x \sim p_\theta(x)$
\end{enumerate}

\subsection{The ELBO}

We want to maximize $\log p_\theta(x)$ but it's intractable. Instead, maximize the Evidence Lower Bound (ELBO):

\begin{mathresult}{ELBO}
\begin{align}
\log p_\theta(x) &\geq \mathcal{L}(\theta, \phi; x) \\
&= \E_{q_\phi(z|x)}[\log p_\theta(x|z)] - D_{KL}(q_\phi(z|x) \| p(z))
\end{align}
\begin{itemize}
    \item First term: \textbf{Reconstruction} (decoder quality)
    \item Second term: \textbf{Regularization} (keep posterior close to prior)
\end{itemize}
\end{mathresult}

\subsection{Reparameterization Trick}

\textbf{Problem}: Can't backprop through sampling $z \sim q_\phi(z|x)$.

\textbf{Solution}: Reparameterize as deterministic function of parameters and noise:
\begin{equation}
z = \mu_\phi(x) + \sigma_\phi(x) \odot \epsilon, \quad \epsilon \sim \mathcal{N}(0, I)
\end{equation}

Now gradients flow through $\mu$ and $\sigma$.

\subsection{VAE Limitations}

\begin{itemize}
    \item \textbf{Blurry samples}: MSE reconstruction encourages averaging
    \item \textbf{Posterior collapse}: Decoder ignores latent, KL goes to zero
    \item \textbf{Limited expressivity}: Gaussian assumptions limit complexity
\end{itemize}

\subsection{Minimax Objective}

\begin{mathresult}{GAN Objective}
\begin{equation}
\min_G \max_D V(D, G) = \E_{x \sim p_{data}}[\log D(x)] + \E_{z \sim p(z)}[\log(1 - D(G(z)))]
\end{equation}
\end{mathresult}

\textbf{Interpretation}:
\begin{itemize}
    \item Discriminator maximizes: correctly classify real/fake
    \item Generator minimizes: fool discriminator
\end{itemize}

\subsection{Training Challenges}

\begin{enumerate}
    \item \textbf{Mode collapse}: Generator produces limited variety
    \item \textbf{Training instability}: Oscillation, divergence
    \item \textbf{Vanishing gradients}: When D is too good
    \item \textbf{Evaluation difficulty}: No likelihood, metrics are proxies
\end{enumerate}

\subsection{WGAN: Wasserstein GAN}

\textbf{Key insight}: Replace JS divergence with Wasserstein distance:
\begin{equation}
W(p_{data}, p_g) = \sup_{\|f\|_L \leq 1} \E_{x \sim p_{data}}[f(x)] - \E_{x \sim p_g}[f(x)]
\end{equation}

\textbf{Benefits}:
\begin{itemize}
    \item Meaningful loss (correlates with quality)
    \item Gradients don't vanish when distributions don't overlap
    \item More stable training
\end{itemize}

\subsection{StyleGAN}

\begin{enumerate}
    \item \textbf{Mapping network}: $z \to w$ (intermediate latent space)
    \item \textbf{AdaIN}: Inject style at each layer via adaptive instance norm
    \item \textbf{Noise injection}: Per-pixel noise for stochastic details
    \item \textbf{Progressive growing}: Start low-res, add layers
\end{enumerate}

\section{Diffusion Models}
\label{sec:diffusion}
%==============================================================================

\subsection{Core Idea}

\begin{enumerate}
    \item \textbf{Forward process}: Gradually add noise until data becomes pure noise
    \item \textbf{Reverse process}: Learn to denoise, step by step
\end{enumerate}

\begin{figure}[H]
\centering
\begin{tikzpicture}[
    node distance=1cm,

]
    \node[img, fill=blue!20] (x0) {$x_0$};

    \node[right=0.5cm of x1] (dots1) {...};
    \node[img, fill=gray!20, right=0.5cm of dots1] (xT) {$x_T$};

    \draw[->] (x1) -- (dots1);
    \draw[->] (dots1) -- (xT);
    
    % Reverse
    \draw[<-, red, thick] (x0.south) -- ++(0,-0.5) -| node[below, font=\tiny, pos=0.25] {$p_\theta(x_{t-1}|x_t)$} (xT.south);
    
    \node[below=1cm of x0, font=\small] {Data};
    \node[below=1cm of xT, font=\small] {Noise};
\end{tikzpicture}
\caption{Diffusion: Forward adds noise, reverse denoises.}
\end{figure}

\subsection{Forward Process}

\begin{mathresult}{Forward Diffusion}
\begin{equation}
q(x_t | x_{t-1}) = \mathcal{N}(x_t; \sqrt{1-\beta_t} x_{t-1}, \beta_t I)
\end{equation}
Closed form for any $t$:
\begin{equation}
q(x_t | x_0) = \mathcal{N}(x_t; \sqrt{\bar{\alpha}_t} x_0, (1-\bar{\alpha}_t) I)
\end{equation}
where $\alpha_t = 1 - \beta_t$ and $\bar{\alpha}_t = \prod_{s=1}^t \alpha_s$.
\end{mathresult}

\subsection{Training Objective}

Train network to predict noise:
\begin{equation}
\mathcal{L} = \E_{t, x_0, \epsilon}\left[\|\epsilon - \epsilon_\theta(x_t, t)\|^2\right]
\end{equation}

where $x_t = \sqrt{\bar{\alpha}_t} x_0 + \sqrt{1-\bar{\alpha}_t} \epsilon$.

\subsection{Sampling}

Starting from $x_T \sim \mathcal{N}(0, I)$, iteratively denoise:
\begin{equation}
x_{t-1} = \frac{1}{\sqrt{\alpha_t}}\left(x_t - \frac{1-\alpha_t}{\sqrt{1-\bar{\alpha}_t}} \epsilon_\theta(x_t, t)\right) + \sigma_t z
\end{equation}

\subsection{Latent Diffusion (Stable Diffusion)}

\textbf{Key insight}: Do diffusion in latent space of pretrained VAE.

\begin{enumerate}
    \item Encode image: $z = E(x)$
    \item Diffusion in latent space: $z_t \to z_{t-1}$
    \item Decode: $x = D(z_0)$
\end{enumerate}

\textbf{Benefits}: 8-16× less computation, enables high-resolution generation.

\subsection{Classifier-Free Guidance}

\begin{equation}
\tilde{\epsilon}_\theta(x_t, c) = \epsilon_\theta(x_t, \emptyset) + w \cdot (\epsilon_\theta(x_t, c) - \epsilon_\theta(x_t, \emptyset))
\end{equation}

\textbf{Intuition}: Amplify the difference between conditional and unconditional predictions.

\textbf{Guidance scale $w$}: Higher = more adherence to condition, lower diversity.

\begin{interviewq}
\textbf{Q: When would you choose VAE vs. GAN vs. Diffusion?}

\textbf{A:}

\textbf{Choose VAE when}:
\begin{itemize}
    \item Need smooth latent space for interpolation/manipulation
    \item Likelihood estimation important
    \item Training stability critical
    \item Representation learning is the goal
\end{itemize}

\textbf{Choose GAN when}:
\begin{itemize}
    \item Fast sampling required
    \item Conditional generation (controlled)
    \item Domain-specific (faces, etc.) with known good architectures
\end{itemize}

\textbf{Choose Diffusion when}:
\begin{itemize}
    \item Maximum sample quality needed
    \item Diversity important (no mode collapse)
    \item Text-to-image or complex conditioning
    \item Can afford slower sampling (or use DDIM)
\end{itemize}

\textbf{Common follow-ups:} ``What about flow-based models?'' (Exact likelihood, invertible, but restrictive architectures.) ``How does Stable Diffusion relate to this?'' (Latent diffusion --- runs diffusion in compressed latent space for efficiency.)
\end{interviewq}

\begin{warningbox}
GAN training is notoriously unstable. Mode collapse (generator produces limited variety) and training divergence are common. If you're choosing between GANs and diffusion for a new project, default to diffusion unless you specifically need fast single-step generation.
\end{warningbox}

\begin{interviewq}
\textbf{Q: How does Stable Diffusion work at a high level? What makes it practical compared to earlier diffusion models?}

\textbf{What the interviewer is testing:} Understanding of latent diffusion and the key insight that made text-to-image practical.

\textbf{A:} Stable Diffusion is a \textit{latent diffusion model}:

\begin{enumerate}
    \item \textbf{VAE encoder} compresses images from pixel space ($512 \times 512 \times 3$) to latent space ($64 \times 64 \times 4$) --- an 8$\times$ spatial compression
    \item \textbf{U-Net} performs the diffusion process (denoising) in this compressed latent space
    \item \textbf{Text conditioning} via cross-attention: CLIP text encoder produces embeddings, which are injected into the U-Net via cross-attention layers
    \item \textbf{VAE decoder} maps the denoised latent back to pixel space
\end{enumerate}

\textbf{Key insight:} Running diffusion in latent space is $\sim$64$\times$ cheaper than pixel space, making high-resolution generation practical.

\textbf{Common follow-ups:} ``What is classifier-free guidance?'' (Train with and without the conditioning, then amplify the difference at inference to increase prompt adherence.) ``How would you fine-tune Stable Diffusion for a specific style?'' (LoRA on the U-Net, DreamBooth for personalization, textual inversion for new concepts.)
\end{interviewq}

\begin{interviewq}
\textbf{Q: Explain the ELBO and why VAEs sometimes produce blurry images.}

\textbf{What the interviewer is testing:} Understanding of generative model training objectives and their practical implications.

\textbf{A:} The ELBO (Evidence Lower Bound) is the VAE training objective:

$\mathcal{L} = \underbrace{-\E_{q(z|x)}[\log p(x|z)]}_{\text{reconstruction}} + \underbrace{\KL(q(z|x) \| p(z))}_{\text{regularization}}$

\textbf{Why blurriness occurs:}
\begin{itemize}
    \item The reconstruction term with MSE loss optimizes for the \textit{expected} output, which is the mean of all possible outputs --- this is inherently blurry
    \item The KL term forces the latent distribution toward a simple Gaussian, limiting the model's expressiveness
    \item GANs avoid this because the discriminator directly evaluates sample quality (not average quality)
\end{itemize}

\textbf{Mitigations:} Use perceptual losses instead of MSE, use adversarial training (VAE-GAN), or use hierarchical VAEs (VQ-VAE, DALL-E) that model discrete latent codes.

\textbf{Common follow-ups:} ``What's VQ-VAE?'' (Uses discrete codebook instead of continuous latent --- avoids posterior collapse, sharper outputs.) ``What's the posterior collapse problem?'' (KL term dominates, encoder ignores input, decoder generates from prior alone.)
\end{interviewq}
